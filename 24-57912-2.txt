function sellProduct() {
    var codeText = document.getElementById("sid").value;
    var qtyText = document.getElementById("sqty").value;
    var customer = document.getElementById("scust").value.trim();
    var msg = document.getElementById("saleMsg");

    if (currentUser.role != "employee") {
        alert("Only an employee can sell at the counter");
        return false;
    }
    if (codeText == "" || qtyText == "") {
        alert("Product code and quantity must be filled out");
        return false;
    }
    if (customer == "") {
        customer = "Walk-in";
    }

    var qty = qtyText * 1;
    if (!(qty > 0)) {
        alert("Quantity must be a number greater than 0");
        return false;
    }

    var found = findProduct(codeText);

    if (found == null) {
        msg.innerHTML = "No product found with code " + codeText.trim().toUpperCase();
        msg.style.color = "red";
        return false;
    }
    if (qty > found.qty) {
        msg.innerHTML = "Not enough stock! Only " + found.qty + " available.";
        msg.style.color = "red";
        return false;
    }

    var total = found.price * qty;
    found.qty -= qty;
    totalRevenue += total;

    sales.push({
        no: sales.length + 1,
        type: "Counter",
        code: found.code,
        product: found.name,
        customer: customer,
        qty: qty,
        total: total,
        soldBy: currentUser.username
    });

    var receipt = "Sold " + qty + " x " + found.name.toUpperCase() + " = " + total + " Tk";
    msg.innerHTML = receipt;
    msg.style.color = "green";
    notify(receipt);

    renderSaleProducts();
    document.getElementById("sid").value = "";
    document.getElementById("sqty").value = "";
    document.getElementById("scust").value = "";
    return false;
}

function findOrder(no) {
    var i;
    for (i = 0; i < orders.length; i++) {
        if (orders[i].no == no) {
            return orders[i];
        }
    }
    return null;
}

function confirmOrder(no) {
    var order = findOrder(no);

    if (currentUser.role != "employee") {
        alert("Only an employee can confirm orders");
        return;
    }
    if (order == null || order.status != "Pending") {
        alert("This order cannot be confirmed");
        return;
    }

    var product = findProduct(order.code);
    if (product == null) {
        alert("This product no longer exists. Please reject the order.");
        return;
    }
    if (order.qty > product.qty) {
        alert("Not enough stock! Only " + product.qty + " available. Add stock first.");
        return;
    }

    product.qty -= order.qty;
    totalRevenue += order.total;
    sales.push({
        no: sales.length + 1,
        type: "Online",
        code: order.code,
        product: order.product,
        customer: order.customerName,
        qty: order.qty,
        total: order.total,
        soldBy: currentUser.username
    });

    order.status = "Completed";
    order.handledBy = currentUser.name;
    notify("Order #" + no + " confirmed");
    renderOrders();
}

function renderShop(filterText) {
    var text = "<table><tr><th>Code</th><th>Name</th><th>Category</th><th>Price (Tk)</th>" +
        "<th>Availability</th><th>Action</th></tr>";
    var i;
    var p;
    var availability;
    var shown = 0;
    var search = filterText.toLowerCase();

    for (i = 0; i < products.length; i++) {
        p = products[i];

        if (search == "" ||
            p.code.toLowerCase().includes(search) ||
            p.name.toLowerCase().includes(search) ||
            p.category.toLowerCase().includes(search)) {

            if (p.qty == 0) {
                availability = "<td class='low'>Out of stock</td>";
            } else if (p.qty <= LOW_STOCK_LIMIT) {
                availability = "<td class='warn'>Only " + p.qty + " left</td>";
            } else {
                availability = "<td class='ok'>Available</td>";
            }

            text += "<tr><td>" + p.code + "</td>" +
                "<td>" + p.name + "</td>" +
                "<td>" + p.category + "</td>" +
                "<td>" + p.price + "</td>" +
                availability +
                "<td><button onclick=\"addToCart('" + p.code + "')\">Add to Cart</button></td></tr>";
            shown++;
        }
    }
    text += "</table>";

    if (shown == 0) {
        text = "<p>No products found.</p>";
    }
    document.getElementById("shopTable").innerHTML = text;
}

function addToCart(codeText) {
    var product = findProduct(codeText);
    var i;
    var line = null;

    if (currentUser == null || currentUser.role != "customer") {
        alert("Only a customer can use the cart");
        return;
    }
    if (product == null) {
        alert("No product found with this code");
        return;
    }
    if (product.qty == 0) {
        alert(product.name + " is out of stock.");
        return;
    }

    for (i = 0; i < cart.length; i++) {
        if (cart[i].code === product.code) {
            line = cart[i];
        }
    }

    if (line == null) {
        cart.push({ code: product.code, name: product.name, price: product.price, qty: 1 });
    } else {
        if (line.qty + 1 > product.qty) {
            alert("Only " + product.qty + " x " + product.name + " in stock.");
            return;
        }
        line.qty++;
    }

    notify(product.name + " added to cart");
    renderCart();
}

function removeFromCart(codeText) {
    var newList = [];
    var i;
    var removedName = "";

    for (i = 0; i < cart.length; i++) {
        if (cart[i].code === codeText) {
            removedName = cart[i].name;
        } else {
            newList.push(cart[i]);
        }
    }

    cart = newList;
    if (removedName != "") {
        notify(removedName + " removed from cart");
    }
    renderCart();
}

function clearCart() {
    cart = [];
    notify("Cart cleared");
    renderCart();
}

function cartTotal() {
    var i;
    var total = 0;
    for (i = 0; i < cart.length; i++) {
        total += cart[i].price * cart[i].qty;
    }
    return total;
}

function placeCartOrder() {
    var msg = document.getElementById("cartMsg");
    var i;
    var product;
    var grandTotal = 0;
    var firstNo = nextOrderNo;

    if (currentUser == null || currentUser.role != "customer") {
        alert("Only a customer can place orders");
        return;
    }
    if (cart.length == 0) {
        alert("Your cart is empty");
        return;
    }

    for (i = 0; i < cart.length; i++) {
        product = findProduct(cart[i].code);
        if (product == null) {
            alert(cart[i].name + " no longer exists. Remove it from the cart.");
            return;
        }
        if (cart[i].qty > product.qty) {
            alert("Not enough stock for " + product.name + ". Only " + product.qty + " available.");
            return;
        }
    }

    for (i = 0; i < cart.length; i++) {
        orders.push({
            no: nextOrderNo,
            username: currentUser.username,
            customerName: currentUser.name,
            code: cart[i].code,
            product: cart[i].name,
            qty: cart[i].qty,
            total: cart[i].price * cart[i].qty,
            status: "Pending",
            handledBy: ""
        });
        grandTotal += cart[i].price * cart[i].qty;
        nextOrderNo++;
    }

    cart = [];
    renderCart();
    msg.innerHTML = "<b>Orders #" + firstNo + " - " + (nextOrderNo - 1) +
        "</b> placed!<br>Total: <b>" + grandTotal + " Tk</b>. " +
        "Waiting for an employee to confirm.";
    msg.style.color = "green";
    notify("Cart order placed, total " + grandTotal + " Tk");
}
